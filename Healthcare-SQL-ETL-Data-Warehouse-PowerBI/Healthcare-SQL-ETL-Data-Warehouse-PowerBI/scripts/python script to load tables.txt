import pandas as pd
from pathlib import Path

folder = Path(
    r"C:\Projects\SQL_ETL\Healthcare-SQL-ETL-Data-Warehouse-PowerBI\data\etl clean"
)

for file in sorted(folder.glob("*.csv")):
    table_name = file.stem

    if table_name.isidentifier():
        globals()[table_name] = pd.read_csv(
            file,
            dtype=str,
            encoding="utf-8-sig",
            keep_default_na=False
        )
